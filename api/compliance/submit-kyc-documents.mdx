---
title: 'Submit KYC Documents'
openapi: 'POST /documentmanagement/attach-documents/'
description: 'Submits KYC documents for customer compliance verification to fulfill outstanding compliance requirements.'
---

This endpoint allows you to submit KYC (Know Your Customer) documents for a customer. Use this after retrieving pending requirements from the [Awaiting Compliance](/api/compliance/awaiting-compliance) endpoint.

## Request Headers

This endpoint requires a valid **Access Token** for authorization.

<ParamField header='X-Auth-Token' type='string' required>
  The **JWT Access Token** obtained from the `/login` or `/refresh-token`
  endpoint.
</ParamField>

<ParamField header='platform' type='string' required default='fincode'>
        Platform identifier. Use `fincode`.
</ParamField>

<ParamField header='uuid' type='string' required default='200'>
        Unique request identifier.
</ParamField>

## Request Body

<ParamField body='customerCode' type='string' required>
  The unique customer identifier for whom the documents are being submitted.
</ParamField>

<ParamField body='attachedDocument' type='array' required>
  List of documents to submit. Each item in the array represents one document.
  <Expandable title='attachedDocument item'>
    <ParamField body='idType' type='string' required>
      The type of KYC document being submitted. Supported values:
      - `PASSPORT` - Valid passport
      - `DRIVING_LICENSE` - Valid driver's license
      - `NATIONAL_ID_CARD` - Government-issued national ID card
      - `VOTE_CARD` - Voter registration card
      - `ADDRESS_PROOF` - Proof of address (e.g. utility bill, bank statement)
      - `SOURCE_OF_INCOME_PROOF` - Proof of income source
      - `CUSTOMER_DECLARATION_FORM` - Customer declaration form
      - `OTHER_OFFICIAL_DOCUMENT` - Any other official document
    </ParamField>
    <ParamField body='file_base64' type='string' required>
      The document file encoded as a Base64 string.
    </ParamField>
    <ParamField body='name' type='string'>
      The filename of the document (e.g. `passport.pdf`).
    </ParamField>
    <ParamField body='associatedRuleId' type='string'>
      The KYC rule ID to associate this document with. Returned as `customer-kyc-record-id` from the awaiting compliance endpoint.
    </ParamField>
    <ParamField body='issueNo' type='string'>
      The document's issue number or reference (e.g. passport number).
    </ParamField>
    <ParamField body='issueCountry' type='string'>
      ISO country code of the issuing country (e.g. `GB`, `IN`).
    </ParamField>
    <ParamField body='issueDate' type='string'>
      Date the document was issued, in `YYYY-MM-DD` format.
    </ParamField>
    <ParamField body='expiryDate' type='string'>
      Date the document expires, in `YYYY-MM-DD` format.
    </ParamField>
    <ParamField body='dob' type='string'>
      Customer's date of birth, in `YYYY-MM-DD` format.
    </ParamField>
  </Expandable>
</ParamField>

## Response

<ResponseField name='status' type='string' default='SUCCESS'>
  Overall status of the API request.
</ResponseField>

<ResponseField name='message' type='string'>
  A descriptive message about the submission result.
</ResponseField>

<ResponseField name='data' type='object'>
  Container for the submission result data.
  <Expandable title='data object'>
    <ResponseField name='accessScope' type='string'>
      The customer's current compliance access scope.
    </ResponseField>
    <ResponseField name='customerStatus' type='string'>
      The customer's current status.
    </ResponseField>
  </Expandable>
</ResponseField>

```javascript Node.js
const axios = require('axios');
const fs = require('fs');

const BASE_URL = 'https://remitjunction.fincode.software/api/v6/services/documentmanagement';

async function submitKycDocument(accessToken, customerCode, ruleId, idType, filePath) {
  const fileBase64 = fs.readFileSync(filePath).toString('base64');
  const fileName = filePath.split('/').pop();

  try {
    const response = await axios.post(
      `${BASE_URL}/attach-documents`,
      {
        customerCode,
        attachedDocument: [
          {
            idType,
            file_base64: fileBase64,
            name: fileName,
            associatedRuleId: ruleId,
          },
        ],
      },
      {
        headers: {
          'X-Auth-Token': accessToken,
          'platform': 'fincode',
          'uuid': '200',
          'Content-Type': 'application/json',
        },
      }
    );

    console.log('Document submitted successfully:', response.data);
    return response.data;
  } catch (error) {
    console.error(
      'Document submission failed:',
      error.response?.data || error.message
    );
    throw error;
  }
}

// Usage Example
const accessToken = 'YOUR_JWT_ACCESS_TOKEN';
submitKycDocument(accessToken, 'CUST123456', 'RULE_ID_FROM_AWAITING_COMPLIANCE', 'ADDRESS_PROOF', './utility_bill.pdf');
```
